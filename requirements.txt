# 跨平台音乐播放器依赖
pillow>=9.0       # 封面/背景图处理 (Web/iPhone 模式也必需)
requests>=2.28    # 酷我/歌词/网易云接口 (Web/iPhone 模式必需)
pygame>=2.3       # 跨平台音频引擎 (macOS/Linux 必装; Windows 可选, 缺失自动回退 winmm MCI)

segno>=1.6        # 手机访问窗口二维码 (可选; 纯 Python)
